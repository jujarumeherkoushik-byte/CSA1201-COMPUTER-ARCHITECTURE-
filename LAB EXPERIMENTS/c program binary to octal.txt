#include <stdio.h>

int main() {
    int binary, decimal = 0, octal = 0;
    int remainder, base = 1, place = 1;

    printf("Enter a binary number: ");
    scanf("%d", &binary);

    // Binary to Decimal
    while (binary > 0) {
        remainder = binary % 10;
        decimal = decimal + remainder * base;
        binary = binary / 10;
        base = base * 2;
    }

    // Decimal to Octal
    while (decimal > 0) {
        remainder = decimal % 8;
        octal = octal + remainder * place;
        decimal = decimal / 8;
        place = place * 10;
    }

    printf("Octal number = %d\n", octal);

    return 0;
}
