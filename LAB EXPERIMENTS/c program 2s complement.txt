#include <stdio.h>

int main() {
    char binary[100];
    int i, carry = 1;

    printf("Enter a binary number: ");
    scanf("%s", binary);

    // 1's complement
    for (i = 0; binary[i] != '\0'; i++) {
        if (binary[i] == '0')
            binary[i] = '1';
        else if (binary[i] == '1')
            binary[i] = '0';
        else {
            printf("Invalid binary number!\n");
            return 1;
        }
    }

    // Add 1
    for (i = 0; binary[i] != '\0'; i++);
    i--;

    while (i >= 0 && carry) {
        if (binary[i] == '0') {
            binary[i] = '1';
            carry = 0;
        } else {
            binary[i] = '0';
        }
        i--;
    }

    printf("2's Complement = %s\n", binary);

    return 0;
}
