#include <stdio.h>

int main() {
    long long binary;
    int decimal = 0, remainder, base = 1;

    printf("Enter a binary number: ");
    scanf("%lld", &binary);

    // Binary to Decimal
    while (binary > 0) {
        remainder = binary % 10;
        decimal += remainder * base;
        binary /= 10;
        base *= 2;
    }

    // Decimal to Hexadecimal
    printf("Hexadecimal number = ");

    if (decimal == 0) {
        printf("0");
    } else {
        char hex[20];
        int i = 0;

        while (decimal > 0) {
            remainder = decimal % 16;

            if (remainder < 10)
                hex[i] = remainder + '0';
            else
                hex[i] = remainder - 10 + 'A';

            decimal /= 16;
            i++;
        }

        while (i > 0)
            printf("%c", hex[--i]);
    }

    printf("\n");

    return 0;
}
